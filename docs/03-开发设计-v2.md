### 4.3 繁简切换模块设计（REQ-010）

#### 4.3.1 converter.js — 繁简转换工具

**文件路径**：`src/client/utils/converter.js`

**职责**：封装 opencc-js 的简体/繁体转换能力，对外提供统一的转换 API。

**API 设计**：

```javascript
// src/client/utils/converter.js
import OpenCC from 'opencc-js';

// 预创建转换器实例（单例模式，避免重复创建）
const s2tConverter = OpenCC.Converter('s', 't'); // 简体 → 繁体
const t2sConverter = OpenCC.Converter('t', 's'); // 繁体 → 简体

/**
 * 简体转繁体
 * @param {string} text - 简体文本
 * @returns {string} 繁体文本
 */
export function toTraditional(text) {
  if (!text) return text;
  return s2tConverter(text);
}

/**
 * 繁体转简体
 * @param {string} text - 繁体文本
 * @returns {string} 简体文本
 */
export function toSimplified(text) {
  if (!text) return text;
  return t2sConverter(text);
}

/**
 * 根据目标模式转换文本
 * @param {string} text - 原始文本
 * @param {'simplified'|'traditional'} mode - 目标模式
 * @returns {string} 转换后的文本
 */
export function convertScript(text, mode) {
  if (!text) return text;
  // 数据库存储的是简体文本，仅在目标为繁体时转换
  if (mode === 'traditional') {
    return toTraditional(text);
  }
  return text;
}
```

**设计说明**：
- opencc-js 的 Converter 实例在模块加载时一次性创建，后续调用为纯函数，性能开销极低（< 1ms/首）
- `convertScript` 函数感知数据源方向：数据库存储简体，仅需 s2t 方向转换
- 不转换 UI 控件文本（按钮、标签等），仅转换诗词内容（标题/作者/正文）

#### 4.3.2 useScriptMode.js — 繁简切换 composable

**文件路径**：`src/client/composables/useScriptMode.js`

**职责**：管理当前繁简模式状态，提供响应式的模式切换能力。

**接口设计**：

```javascript
// src/client/composables/useScriptMode.js
import { ref, computed } from 'vue';
import { convertScript } from '../utils/converter.js';

const STORAGE_KEY = 'scriptMode';

// 模块级单例状态（所有组件共享同一状态）
const currentMode = ref('simplified');

// 初始化：从 sessionStorage 恢复状态
function initMode() {
  try {
    const saved = sessionStorage.getItem(STORAGE_KEY);
    if (saved === 'traditional' || saved === 'simplified') {
      currentMode.value = saved;
    }
  } catch {
    // sessionStorage 不可用时忽略，保持默认简体
  }
}
initMode();

/**
 * 繁简切换组合式函数
 * @returns {Object} 繁简切换相关状态与方法
 */
export function useScriptMode() {
  /** 当前是否为繁体模式 */
  const isTraditional = computed(() => currentMode.value === 'traditional');

  /** 切换繁简模式 */
  function toggleScriptMode() {
    currentMode.value = isTraditional.value ? 'simplified' : 'traditional';
    try {
      sessionStorage.setItem(STORAGE_KEY, currentMode.value);
    } catch {
      // sessionStorage 不可用时不阻断切换
    }
  }

  /**
   * 根据当前模式转换文本
   * @param {string} text - 原始文本（简体）
   * @returns {string} 转换后的文本
   */
  function convertText(text) {
    return convertScript(text, currentMode.value);
  }

  return {
    currentMode,       // Ref<string> 'simplified' | 'traditional'
    isTraditional,     // ComputedRef<boolean>
    toggleScriptMode,  // () => void
    convertText,       // (text: string) => string
  };
}
```

**设计说明**：
- 使用模块级 `ref` 作为单例状态，所有调用 `useScriptMode()` 的组件共享同一响应式引用，切换时全局同步更新
- `sessionStorage` 持久化仅用于刷新恢复，不作为跨页面通信手段（跨页面通过 Vue 响应式系统自动同步）
- `convertText` 方法封装了转换方向判断，组件调用时无需关心当前模式

#### 4.3.3 受影响组件的改造方式

| 组件 | 改造内容 |
|---|---|
| `nav-bar.vue` | 新增繁简切换按钮，调用 `useScriptMode()` 获取 `isTraditional` 和 `toggleScriptMode`，按钮文字在"繁"/"简"之间切换 |
| `poem-card.vue` | 调用 `useScriptMode()` 获取 `convertText`，标题、作者、正文摘要均通过 `convertText()` 转换后渲染 |

### 4.4 收藏模块设计（REQ-013）

#### 4.4.1 storage.js — localStorage 封装

**文件路径**：`src/client/utils/storage.js`

**职责**：封装 localStorage 的读写操作，提供异常安全的存储接口。

**API 设计**：

```javascript
// src/client/utils/storage.js

const FAVORITES_KEY = 'poetry_favorites';
const CURRENT_VERSION = 1;

/**
 * 检测 localStorage 是否可用
 * @returns {boolean}
 */
export function isStorageAvailable() {
  try {
    const testKey = '__storage_test__';
    localStorage.setItem(testKey, '1');
    localStorage.removeItem(testKey);
    return true;
  } catch {
    return false;
  }
}

/**
 * 读取收藏数据
 * @returns {{ version: number, items: Array<FavoriteItem> }}
 */
export function loadFavorites() {
  try {
    const raw = localStorage.getItem(FAVORITES_KEY);
    if (!raw) return { version: CURRENT_VERSION, items: [] };
    const data = JSON.parse(raw);
    // 版本检查，预留迁移能力
    if (data.version !== CURRENT_VERSION) {
      return migrateFavorites(data);
    }
    return data;
  } catch {
    // JSON 解析失败，数据损坏，返回空结构
    return { version: CURRENT_VERSION, items: [] };
  }
}

/**
 * 保存收藏数据
 * @param {{ version: number, items: Array<FavoriteItem> }} data
 * @returns {boolean} 是否保存成功
 */
export function saveFavorites(data) {
  try {
    localStorage.setItem(FAVORITES_KEY, JSON.stringify(data));
    return true;
  } catch (e) {
    // 配额溢出等异常
    console.warn('[Storage] 保存收藏失败:', e.message);
    return false;
  }
}

/**
 * 数据结构迁移（预留）
 * @param {Object} oldData
 * @returns {{ version: number, items: Array }}
 */
function migrateFavorites(oldData) {
  // 当前 version=1，无需迁移，直接返回空
  return { version: CURRENT_VERSION, items: [] };
}
```

**localStorage 数据结构**：

```json
{
  "version": 1,
  "items": [
    {
      "id": "tang-001",
      "title": "静夜思",
      "author": "李白",
      "dynasty": "唐",
      "category": "五言绝句",
      "addedAt": "2026-10-01T12:00:00.000Z"
    }
  ]
}
```

| 字段 | 类型 | 说明 |
|---|---|---|
| `version` | number | 数据结构版本号，预留升级能力 |
| `items` | Array | 收藏列表，按 `addedAt` 降序排列 |
| `items[].id` | string | 诗词唯一 ID（与 poems 表 id 一致） |
| `items[].title` | string | 诗词标题 |
| `items[].author` | string | 作者姓名 |
| `items[].dynasty` | string | 朝代名称 |
| `items[].category` | string | 分类（诗体/词牌） |
| `items[].addedAt` | string | 收藏时间（ISO 8601 格式） |

#### 4.4.2 useFavorites.js — 收藏 composable

**文件路径**：`src/client/composables/useFavorites.js`

**职责**：封装收藏的增删查改业务逻辑，与存储层解耦。

**接口设计**：

```javascript
// src/client/composables/useFavorites.js
import { ref, computed } from 'vue';
import { loadFavorites, saveFavorites, isStorageAvailable } from '../utils/storage.js';

// 模块级单例状态
const favorites = ref([]);
const initialized = ref(false);

/**
 * 收藏功能组合式函数
 * @returns {Object} 收藏相关状态与方法
 */
export function useFavorites() {
  // 惰性初始化：首次调用时从 localStorage 加载
  if (!initialized.value) {
    const data = loadFavorites();
    favorites.value = data.items;
    initialized.value = true;
  }

  /** 收藏列表（按 addedAt 降序） */
  const favoriteList = computed(() =>
    [...favorites.value].sort((a, b) => new Date(b.addedAt) - new Date(a.addedAt))
  );

  /** 收藏数量 */
  const favoriteCount = computed(() => favorites.value.length);

  /** localStorage 是否可用 */
  const storageAvailable = isStorageAvailable();

  /**
   * 判断某首诗词是否已收藏
   * @param {string} poemId - 诗词 ID
   * @returns {boolean}
   */
  function isFavorite(poemId) {
    return favorites.value.some(item => item.id === poemId);
  }

  /**
   * 收藏一首诗词
   * @param {{ id: string, title: string, author: string, dynasty: string, category: string }} poem
   * @returns {boolean} 操作是否成功
   */
  function addFavorite(poem) {
    if (isFavorite(poem.id)) return true; // 已收藏，幂等
    const item = {
      id: poem.id,
      title: poem.title,
      author: poem.author,
      dynasty: poem.dynasty,
      category: poem.category || '',
      addedAt: new Date().toISOString(),
    };
    favorites.value = [...favorites.value, item];
    return saveFavorites({ version: 1, items: favorites.value });
  }

  /**
   * 取消收藏
   * @param {string} poemId - 诗词 ID
   * @returns {boolean} 操作是否成功
   */
  function removeFavorite(poemId) {
    favorites.value = favorites.value.filter(item => item.id !== poemId);
    return saveFavorites({ version: 1, items: favorites.value });
  }

  /**
   * 切换收藏状态
   * @param {{ id: string, title: string, author: string, dynasty: string, category: string }} poem
   * @returns {boolean} 操作后是否为已收藏状态
   */
  function toggleFavorite(poem) {
    if (isFavorite(poem.id)) {
      removeFavorite(poem.id);
      return false;
    } else {
      addFavorite(poem);
      return true;
    }
  }

  return {
    favoriteList,      // ComputedRef<Array>
    favoriteCount,     // ComputedRef<number>
    storageAvailable,  // boolean
    isFavorite,        // (poemId: string) => boolean
    addFavorite,       // (poem) => boolean
    removeFavorite,    // (poemId: string) => boolean
    toggleFavorite,    // (poem) => boolean
  };
}
```

**跨标签页同步**：

在 `useFavorites` 初始化时监听 `storage` 事件，当其他标签页修改收藏数据时自动同步：

```javascript
// 在 useFavorites 初始化逻辑中添加
window.addEventListener('storage', (e) => {
  if (e.key === 'poetry_favorites') {
    const data = loadFavorites();
    favorites.value = data.items;
  }
});
```

### 4.6 搜索高亮修复设计（REQ-011）

#### 4.6.1 问题根因

一期 `poem-card.vue` 的高亮函数 `highlightedSummary` 直接使用 `props.highlightKeyword` 构建正则表达式匹配 `poem.summary`。而 `search-view.vue` 传递的 `highlightKeyword` 来自 `route.query.q`（用户输入的简体关键词），当数据库正文为繁体时匹配失败。

后端搜索 API 已返回 `originalKeyword` 字段（用户原始输入），但前端未使用。

#### 4.6.2 修复方案

**search-view.vue 变更**：

```javascript
// 一期（有问题）：
// const highlightKeyword = computed(() => route.query.q);

// 二期（修复）：从 API 响应中读取 originalKeyword
const originalKeyword = ref('');

// 在搜索 API 返回后赋值
async function doSearch() {
  const res = await api.searchPoems(route.query.q, page, pageSize);
  originalKeyword.value = res.data.originalKeyword || route.query.q;
  // ...
}
```

**poem-card.vue 高亮函数增强**：

```javascript
// 一期签名：
// const highlightedSummary = computed(() => { ... })

// 二期签名：新增繁简匹配能力
import { toTraditional } from '../utils/converter.js';

const highlightedSummary = computed(() => {
  if (!props.highlightKeyword || !props.poem.summary) return props.poem.summary;
  const keyword = props.highlightKeyword;
  const text = props.poem.summary;

  // 第一步：使用原始关键词尝试匹配
  let result = highlightText(text, keyword);
  if (result !== text) return result;

  // 第二步：原始关键词未匹配，尝试将关键词转为繁体后再匹配
  const traditionalKeyword = toTraditional(keyword);
  if (traditionalKeyword !== keyword) {
    result = highlightText(text, traditionalKeyword);
    if (result !== text) return result;
  }

  // 第三步：仍未匹配，尝试将关键词转为简体后匹配
  // （覆盖正文为简体但关键词含特殊字符的情况）
  return text;
});

/**
 * 安全高亮文本中的关键词
 * @param {string} text - 原始文本
 * @param {string} keyword - 要高亮的关键词
 * @returns {string} 包含 <mark> 标签的 HTML 字符串
 */
function highlightText(text, keyword) {
  const escaped = keyword.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const regex = new RegExp(`(${escaped})`, 'gi');
  if (!regex.test(text)) return text;
  return text.replace(regex, '<mark class="highlight">$1</mark>');
}
```

**设计说明**：
- 高亮使用 `<mark class="highlight">` 标签包裹，通过 CSS 变量 `--color-highlight` 控制背景色
- 正则特殊字符通过 `escapeRegExp` 转义，避免注入
- 三级匹配策略：原始关键词 → 繁体转换 → 放弃高亮（返回原文）
- `v-html` 渲染高亮 HTML 时，仅对 `poem.summary`（后端返回的纯文本）做替换，不处理用户输入，XSS 风险可控

### 5.2 app.js 导出方式调整

**现状**：一期 `app.js` 在文件末尾同时执行了 `app.listen()` 和 `export default app`。`app.listen()` 在模块导入时即执行，导致 Supertest 测试时会自动启动 HTTP 监听，与测试框架冲突。

**调整方案**：将 `app.listen()` 提取到独立的启动文件中，`app.js` 仅负责创建和导出 app 实例。

```javascript
// src/server/app.js（调整后）

// 移除 app.listen()，仅导出 app 实例
export default app;

import app from './app.js';

```

**影响范围**：仅调整启动入口，不影响任何业务逻辑。

### 5.3 数据库路径优化

**现状**：一期 `database.js` 使用相对路径（向上 4 级 `../../../../data/poetry.db`）定位数据库文件，路径脆弱，在不同工作目录下启动时可能找不到数据库。

## 九、风险与缓解

### 9.1 技术风险

| 编号 | 风险描述 | 等级 | 影响范围 | 缓解措施 |
|---|---|---|---|---|
| R9 | opencc-js 前端引入导致打包体积增大约 300KB（字典文件） | 中 | 首屏加载性能 | (1) 使用 Vite 动态导入 `import('opencc-js')` 按需加载，避免打入主 bundle；(2) converter.js 仅在首次调用时触发字典加载，后续调用为纯函数；(3) 通过 gzip 压缩后字典约 100KB，对首屏影响可控 |
| R10 | 繁简切换与搜索高亮联动复杂，切换后高亮可能失效 | 中 | REQ-010/011 功能正确性 | (1) poem-card 高亮函数内置三级匹配策略（原始→繁体→放弃），确保切换后仍能正确高亮；(2) 高亮关键词来源统一使用 originalKeyword，不受繁简切换状态影响；(3) 开发阶段编写专项验证：切换字体后检查搜索结果高亮 |
| R11 | 导航栏重构（汉堡菜单）改动面大，可能导致一期功能回归 | 中 | AC14 回归无破坏 | (1) 采用渐进策略：先完成 CSS 适配（不改交互），再处理汉堡菜单；(2) 导航栏桌面端样式通过媒体查询保护（`@media (min-width: 768px)` 维持原行为）；(3) 每个组件改动后手动回归一期功能 |
| R12 | localStorage 在隐私模式/无痕模式下不可用 | 低 | REQ-013 收藏功能 | (1) `storage.js` 的 `isStorageAvailable()` 在初始化时探测可用性；(2) 不可用时收藏按钮置灰，点击提示"当前浏览器不支持收藏"；(3) 不阻断其他功能使用 |
| R13 | 测试依赖 poetry.db，CI 或新环境无数据库时测试全部失败 | 低 | REQ-014 自动化测试 | (1) `setup.js` 检测数据库文件是否存在，不存在时 `describe.skip()` 跳过全部用例并输出友好提示；(2) 后续可考虑提供小型 test fixture 数据库 |

### 9.2 回归保障

| 措施 | 说明 |
|---|---|
| 后端零侵入 | 不修改 service/dao/routes 代码，从根源杜绝后端回归 |
| 前端增量修改 | 已有组件仅在原有逻辑基础上追加调用，不改变原有数据流 |
| CSS 媒体查询隔离 | 移动端样式通过 `@media` 包裹，不影响桌面端默认样式 |
| 自动化测试兜底 | 35 个 API 测试用例覆盖 7 个端点，回归时可快速验证 |
